import { APP_BASE_HREF } from '@angular/common';
import { CommonEngine } from '@angular/ssr/node';
import express from 'express';
import { fileURLToPath } from 'node:url';
import { dirname, join, resolve } from 'node:path';
import { existsSync } from 'node:fs';
import bootstrap from './src/main.server';

// The Express app is exported so that it can be used by serverless Functions.
export function app(): express.Express {
  const server = express();
  const serverDistFolder = dirname(fileURLToPath(import.meta.url));
  const browserDistFolder = resolve(serverDistFolder, '../browser');
  const indexHtml = join(serverDistFolder, 'index.server.html');
  const apiBaseUrl = 'https://api.fanspi.com';

  // CONFIGURACIÓN AQUÍ: Se pasan los dominios permitidos al instanciar el motor
  const commonEngine = new CommonEngine({
    allowedHosts: ['fanspi.com', 'www.fanspi.com', 'localhost']
  });

  server.set('view engine', 'html');
  server.set('views', browserDistFolder);

  // Example Express Rest API endpoints
  server.use('/api', async (req, res) => {
    try {
      const targetUrl = `${apiBaseUrl}${req.originalUrl}`;

      // Copiar cabeceras relevantes evitando conflictos de host o longitud
      const headers: Record<string, string> = {};
      for (const [key, value] of Object.entries(req.headers)) {
        if (value && !['host', 'content-length', 'connection'].includes(key.toLowerCase())) {
          headers[key] = Array.isArray(value) ? value.join(', ') : value;
        }
      }

      const hasBody = !['GET', 'HEAD'].includes(req.method.toUpperCase());
      const fetchOptions: RequestInit = {
        method: req.method,
        headers,
        body: hasBody ? req : undefined,
        // Necesario en Node fetch cuando se envía un stream como body
        ...(hasBody ? { duplex: 'half' } : {}),
      } as RequestInit;

      const upstreamResponse = await fetch(targetUrl, fetchOptions);
      res.status(upstreamResponse.status);

      upstreamResponse.headers.forEach((val, key) => {
        if (!['content-encoding', 'transfer-encoding', 'connection'].includes(key.toLowerCase())) {
          res.setHeader(key, val);
        }
      });

      const arrayBuffer = await upstreamResponse.arrayBuffer();
      res.send(Buffer.from(arrayBuffer));
    } catch (error) {
      // Registrar error del proxy sin exponer detalles internos al cliente
      console.error('SSR API proxy error:', error);
      res.status(502).json({ message: 'Bad Gateway' });
    }
  });

  // Serve static files from /browser
  // index: false para evitar que express.static sirva un index.html viejo o estático cuando se visita la raíz '/'
  server.get('**', express.static(browserDistFolder, {
    maxAge: '1y',
    index: false,
    setHeaders: (res, filePath) => {
      // Manifiesto, service worker y archivos HTML nunca deben cachearse agresivamente
      if (filePath.endsWith('.html') || filePath.endsWith('ngsw.json') || filePath.endsWith('manifest.webmanifest')) {
        res.setHeader('Cache-Control', 'no-cache, no-store, must-revalidate');
        res.setHeader('Pragma', 'no-cache');
        res.setHeader('Expires', '0');
      }
    }
  }));

  // Evitar que peticiones a archivos con extensión no encontrados (.js, .css, .ico, etc.) caigan en el motor SSR
  server.get('*.*', (req, res) => {
    res.status(404).type('text/plain').send('Resource not found');
  });

  // All regular routes use the Angular engine
  server.get('**', (req, res, next) => {
    const { protocol, originalUrl, baseUrl, headers } = req;

    commonEngine
      .render({
        bootstrap,
        documentFilePath: indexHtml,
        url: `${protocol}://${headers.host}${originalUrl}`,
        publicPath: browserDistFolder,
        providers: [{ provide: APP_BASE_HREF, useValue: baseUrl }],
      })
      .then((html) => res.send(html))
      .catch((err) => {
        console.error('SSR render error for:', originalUrl, err);
        // Fallback resiliente: enviar el index cliente para que la app cargue en el navegador (CSR)
        const clientIndex = existsSync(join(browserDistFolder, 'index.csr.html'))
          ? join(browserDistFolder, 'index.csr.html')
          : join(browserDistFolder, 'index.html');

        res.sendFile(clientIndex, (sendErr) => {
          if (sendErr) next(err);
        });
      });
  });

  return server;
}

function run(): void {
  const port = process.env['PORT'] || 4000;

  // Start up the Node server
  const server = app();
  server.listen(port, () => {
    console.log(`Node Express server listening on http://localhost:${port}`);
  });
}

run();
